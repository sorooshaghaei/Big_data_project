\section{Appendix: Core Concepts \& Prerequisites}

% --- PART 1: THE PROBLEM BASICS ---

\begin{frame}{1.1 What is Time-Series Data?}
    \textbf{Time-Series} is data collected over time, in chronological order. 

    The paper frames ridership prediction as a time-series forecasting problem. Imagine counting people at a door every 15 minutes.
    \begin{itemize}
        \item $t$ represents the \textit{current} time interval (e.g., 10:00 AM).
        \item $t-1$ is the \textit{previous} interval (9:45 AM).
        \item $t+1$ is the \textit{next} interval (10:15 AM).
    \end{itemize}

    \vspace{0.2cm}
    When the paper says it uses $n=4$ historical steps to predict $m=4$ future steps, it means:
    \textit{"I will look at the last 60 minutes (four 15-min chunks) to guess the next 60 minutes."}
\end{frame}

\begin{frame}{1.2 The Problem: Time-Series Forecasting}
    \textbf{What are we trying to do?}
    Predict how many passengers will enter and exit a metro station in the future based on past data.
    
    \vspace{0.3cm}
    \textbf{The Notation (The Math):}
    \begin{itemize}
        \item Past data sequence (length $n$): $(D_{t-n+1}^{N},D_{t-n+2}^{N},...,D_{t}^{N})$.
        \item Future data sequence (length $m$): $(\hat{D}_{t+1}^{N},\hat{D}_{t+2}^{N},...,\hat{D}_{t+m}^{N})$.
        \item $N$ is the total number of stations.
        \item $D \in \mathbb{R}^{2}$ means the data has two values:
        \begin{itemize}
            \item \textbf{Inflow Count:} People entering the station to ride.
            \item \textbf{Outflow Count:} People leaving the station into the city.
        \end{itemize}
    \end{itemize}
\end{frame}

% --- PART 2: NEURAL NETWORK FOUNDATIONS ---

\begin{frame}{2.1 What is a Neural Network?}
    \textbf{Goal:} Learn a function that maps inputs to outputs.
    
    Example in this paper:
    \[
    \text{Past ridership data} \rightarrow \text{Future ridership prediction}
    \]
    
    A neural network is a stack of mathematical functions $y = f(x)$, where:
    \begin{itemize}
        \item $x$ = input data
        \item $f$ = learned function
        \item $y$ = predicted output
    \end{itemize}
    
    Instead of manually designing the function, the network \textbf{learns it from data}.
\end{frame}

\begin{frame}{2.2 Weights and Parameters}
    Neural networks learn by adjusting internal numbers called \textbf{parameters} or \textbf{weights}.
    
    \vspace{0.3cm}
    Example of a simple neuron:
    \[
    y = w_1x_1 + w_2x_2 + b
    \]
    Where:
    \begin{itemize}
        \item $x_1, x_2$ = input features
        \item $w_1, w_2$ = weights (parameters to learn)
        \item $b$ = bias
    \end{itemize}
    
    \vspace{0.3cm}
    During training, the model changes these weights so predictions become more accurate.
\end{frame}

\begin{frame}{2.3 Forward Pass: Making a Prediction}
    Training a neural network starts with a \textbf{forward pass}.
    
    Steps:
    \begin{enumerate}
        \item Input data enters the network.
        \item Each layer applies a mathematical transformation.
        \item The network produces a prediction.
    \end{enumerate}
    
    Example:
    \[
    \text{Past metro data} \rightarrow \text{Neural network} \rightarrow \hat{y} \text{ (predicted ridership)}
    \]
    
    At this stage, the prediction is usually wrong because the weights are not yet trained.
\end{frame}

\begin{frame}{2.4 Loss Function: Measuring Error}
    After making a prediction, we measure how wrong it is using a \textbf{loss function}.
    
    \vspace{0.3cm}
    Example used in the paper: L2 loss
    \[
    L = (\hat{y} - y)^2
    \]
    Where:
    \begin{itemize}
        \item $y$ = true historical value
        \item $\hat{y}$ = AI's predicted value
    \end{itemize}
    
    \vspace{0.3cm}
    Goal of training: \textbf{Minimize the loss.} The smaller the loss, the better the prediction.
\end{frame}

% --- PART 3: TRAINING MECHANICS (HOW IT LEARNS) ---

\begin{frame}[fragile]{3.1 What is Backpropagation?}
    \small
    \textbf{The Concept:}
    If the "Forward Pass" is guessing, \textbf{Backpropagation} is learning from the mistake. It sends the error backwards to update the weights.

    \begin{verbatim}
    1. FORWARD (Guessing):
       Data ---> [Weight A] ---> [Weight B] ---> Guess (100 people)
                                                 Real Answer (150) 
                                                 ERROR = 50
    
    2. BACKWARD (Learning via Backpropagation):
       Data <--- [Update A] <--- [Update B] <--- ERROR = 50
                 (Blame A)       (Blame B)
    \end{verbatim}

    \begin{itemize}
        \item It uses the \textbf{Chain Rule} from calculus to calculate "Gradients".
        \item It asks: \textit{"How much did Weight B contribute to the error? How about A?"}
        \item The Optimizer uses this "blame" to tweak the weights.
    \end{itemize}
\end{frame}

\begin{frame}{3.2 Gradients and Backpropagation (The Math)}
    To improve predictions, the network adjusts its weights using \textbf{gradients}.
    
    A gradient tells us:
    \begin{quote}
    "If we slightly change this weight, how will the loss change?"
    \end{quote}
    
    Mathematically:
    \[
    \frac{\partial L}{\partial w}
    \]
    
    \textbf{The exact Backpropagation sequence:}
    1. Compute prediction (Forward Pass) $\rightarrow$ 2. Measure loss $\rightarrow$ 3. Compute gradients (Backward Pass) $\rightarrow$ 4. Update weights. \\
    This repeats thousands of times.
\end{frame}

\begin{frame}{3.3 How AI Learns: The "Mountain" Analogy}
    Imagine the AI is standing at the top of a mountain, blindfolded. 
    \begin{itemize}
        \item \textbf{Top of the mountain:} 100\% wrong (high error/loss).
        \item \textbf{Bottom of the valley:} 100\% right (zero error).
    \end{itemize}
    
    The AI wants to walk down to the bottom. 
    \begin{itemize}
        \item The \textbf{Loss Function} tells the AI how high up the mountain it currently is.
        \item The \textbf{Optimizer} is the strategy the AI uses to feel the ground with its foot and decide which direction to step next to go downhill.
    \end{itemize}
\end{frame}

\begin{frame}{3.4 What is the Adam Optimizer?}
    An \textbf{Optimizer} is the mathematical engine that updates the weights based on the gradients. 
    
    \textbf{Adam} (Adaptive Moment Estimation) is a popular, smart optimizer.
    \begin{itemize}
        \item Instead of taking the same size step every time, Adam adapts.
        \item If the mountain is steep, Adam takes big, fast steps. 
        \item If the AI feels it is getting close to the bottom (the right answer), Adam automatically takes tiny, careful steps so it doesn't overshoot the target.
    \end{itemize}
\end{frame}

\begin{frame}{3.5 Epochs and The Training Loop}
    \textbf{What is an Epoch?}
    Think of the historical metro dataset as a giant stack of flashcards. An \textbf{Epoch} is exactly ONE full pass through the entire stack. This model trained for \textbf{250 epochs} (it reviewed the history 250 times).

    \vspace{0.2cm}
    \textbf{The Full Training Loop:}
    \begin{enumerate}
        \item \textbf{Guess:} AI looks at 60 mins of past data, guesses next 60 mins.
        \item \textbf{Check Error:} Compares guess to the real answer (L2-loss).
        \item \textbf{Learn:} Backpropagation finds gradients; Adam tweaks weights.
        \item \textbf{Repeat:} Do this for all data (1 Epoch). Repeat 250 times!
    \end{enumerate}
\end{frame}

% --- PART 4: VANISHING GRADIENTS & ACTIVATION FUNCTIONS ---

\begin{frame}[fragile]{4.1 How Gradients Flow \& Why They Vanish}
    \small
    During training, the error signal travels backward. Each layer multiplies the gradient by a fraction. If many layers are stacked, the gradient shrinks rapidly.

    \begin{verbatim}
    Layer 4 gradient = 1.0 × 0.5 = 0.5
    Layer 3 gradient = 0.5 × 0.5 = 0.25
    Layer 2 gradient = 0.25 × 0.5 = 0.125
    Layer 1 gradient = 0.125 × 0.5 = 0.0625  <-- Almost Zero!
    \end{verbatim}

    \textbf{The Vanishing Gradient Problem:}
    When gradients become extremely small near the input layers:
    \begin{itemize}
        \item Weight updates become tiny.
        \item Early layers stop learning entirely.
        \item Training becomes very slow or stalls.
    \end{itemize}
\end{frame}

\begin{frame}{4.2 Activation Functions in Neural Networks}
    \textbf{Motivation:}
    Neural networks naturally apply simple linear transformations (straight lines). If only linear operations were used, the network could never learn complex patterns (like sudden crowd spikes at a metro).

    \vspace{0.3cm}
    \textbf{Solution: Activation Functions}
    \begin{itemize}
        \item Activation functions introduce \textbf{nonlinearity}.
        \item They act as "gates" deciding if a neuron should fire or not.
        \item Without them, deep networks are useless; with them, networks can approximate almost any complex real-world data curve.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{4.3 Sigmoid vs. ReLU}
    \small
    Older models used the \textbf{Sigmoid} function (an S-curve between 0 and 1). At the flat edges of the curve, the slope is nearly 0. This caused massive Vanishing Gradients.

    \textbf{The Fix: ReLU (Rectified Linear Unit)}
    $$ReLU(x) = \max(0, x)$$
    \begin{itemize}
        \item If $x \leq 0$, output is 0. If $x > 0$, output is $x$.
        \item Because the slope for positive numbers is a constant straight line, gradients flow perfectly without shrinking.
    \end{itemize}

    \vspace{-0.2cm}
    \begin{verbatim}
    ReLU    |       /     Gradient is constant for positive inputs.
            |      /      This stops gradients from vanishing!
            |_____/
            +-------------
    \end{verbatim}
\end{frame}

% --- PART 5: GRAPH THEORY BASICS ---

\begin{frame}[fragile]{5.1 Graph Theory: Mapping the Metro}
    In Big Data, we represent networks using \textbf{Graphs}. This paper uses THREE different graphs to give the AI the full picture:
    
    \begin{verbatim}
    1. Physical Graph     2. Semantic Graph     3. Correlation Graph
    (Physical Tracks)     (Similar Behaviors)   (Actual Travel Routes)
    
       A --- B                A       B            A ===== B
       |     |                |       |             \     /
       C --- D                C       D              C   D
    (Who is next door?)   (Who is busy at the   (Where do people
                           same time?)           actually travel?)
    \end{verbatim}
    
    An \textbf{Edge Weight} is a number showing how strongly two stations are connected in these specific views.
\end{frame}

\begin{frame}{5.2 Physical ($M_p$) \& Semantic ($M_s$) Graphs}
    \small
    \textbf{Physical Edge Weight ($M_p$):} 
    Are these stations directly connected by a train track?
    \begin{itemize}
        \item Base score is 1 if connected, 0 if not. Formula standardizes it to fractions.
        \item \textit{Why it matters:} Captures direct crowd spillover to the next station.
    \end{itemize}

    \vspace{0.3cm}
    \textbf{Semantic Similarity ($M_s$) \& DTW:}
    Do these stations have the same "vibe"? (e.g., two business districts far apart that both peak at 5:00 PM).
    \begin{itemize}
        \item Uses \textbf{Dynamic Time Warping (DTW)}: An algorithm that matches similar time-series curves even if one is slightly delayed.
        \item \textit{Why it matters:} Connects stations based on human behavior, not geography.
    \end{itemize}
\end{frame}

\begin{frame}{5.3 Correlation Matrix ($M_c$)}
    \textbf{Correlation Matrix ($M_c$):}
    Are people actually riding the train directly from Station A to Station B? 
    
    \begin{itemize}
        \item The physical graph shows the tracks.
        \item The semantic graph shows the schedule/behavior.
        \item \textbf{The correlation graph shows the actual humans.} It counts total travels starting at A and ending at B.
    \end{itemize}
    
    \textbf{The Math:} 
    Takes trips between A and B ($R_c$) and divides by total trips in network.
    $$M_{c}(a,b)=\frac{R_{c}(a,b)}{\sum_{k=1}^{N}R_{c}(a,k)}$$
    
    \textit{Why it matters:} Directly maps where crowds are flowing in real-time.
\end{frame}

% --- PART 6: TRANSFORMERS & ATTENTION ---

\begin{frame}{6.1 RNNs vs. Transformers}
    \textbf{The Old Way: RNNs (Recurrent Neural Networks)}
    \begin{itemize}
        \item RNNs read data chronologically, step-by-step.
        \item \textit{The Flaw:} They struggle to remember old information. They suffer from vanishing gradients over long time horizons.
    \end{itemize}

    \vspace{0.3cm}
    \textbf{The New Way: Transformers}
    \begin{itemize}
        \item Transformers don't read step-by-step. They look at the entire sequence at once.
        \item They use \textbf{Self-Attention} to score which past time steps are the most important for predicting the future.
    \end{itemize}
\end{frame}

\begin{frame}{6.2 Embeddings: Converting Data into Vectors}
    \textbf{Problem:} Neural networks cannot process raw concepts, words, or raw time steps easily.
    
    \textbf{Solution:} \textbf{Embeddings} transform each time-step token into a dense vector of real numbers.
    \begin{itemize}
        \item Similar data patterns obtain similar vectors in the mathematical space.
        \item These embeddings become the starting input used to compute \textbf{Query (Q)}, \textbf{Key (K)}, and \textbf{Value (V)} vectors in the Transformer.
    \end{itemize}
\end{frame}

\begin{frame}{6.3 Unpacking "Self-Attention"}
    \textit{Analogy: Reading a sentence.}
    If you read: "The \textbf{bank} of the \textbf{river} was muddy."
    To understand the word "bank", your brain pays \textit{attention} to the word "river" (not a money bank). 
    
    \vspace{0.2cm}
    \textbf{Step 1: Q, K, V Vectors}
    From the embedding, the network creates 3 vectors:
    \begin{itemize}
        \item \textbf{Query (Q)} – what the token is looking for.
        \item \textbf{Key (K)} – what the token represents.
        \item \textbf{Value (V)} – the actual information the token carries.
    \end{itemize}

    \textbf{Step 2: Attention Scores}
    The model computes similarity by multiplying Query and Key ($Q \cdot K$). Higher scores mean the model should pay more attention to that time step.
\end{frame}

\begin{frame}[fragile]{6.4 What is Softmax?}
    \small
    \textbf{The Concept:} 
    Softmax is a mathematical function that takes raw, random Attention scores and converts them into \textbf{probabilities} that sum to 1.0 (100\%).
    
    \begin{verbatim}
    Raw Attention Scores        Softmax Math         Final Probabilities
    
    Time 4:30 PM:  3.2   --->  e^(3.2) / Sum   --->   84%  (Focus here!)
    Time 4:45 PM:  1.0   --->  e^(1.0) / Sum   --->   11%  (A little)
    Time 4:00 PM: -0.5   --->  e^(-0.5) / Sum  --->    5%  (Ignore)
    
    TOTAL:         3.7                               100% 
    \end{verbatim}

    \textbf{Step 3: Final Output}
    The Softmax probabilities are multiplied by the \textbf{Value (V)} vectors. The AI successfully filters out the noise and focuses 84\% of its computing power on the 4:30 PM data spike!
\end{frame}

\begin{frame}{6.5 Self-Attention Mechanism (The Math)}
    \textbf{Computation Summary:}
    \[
    \text{Attention}(Q,K,V) = softmax\left(\frac{QK^T}{\sqrt{d_k}}\right)V
    \]
    \begin{itemize}
        \item $Q$ = Query matrix, $K$ = Key matrix, $V$ = Value matrix.
        \item The dot product $QK^T$ measures similarity between time steps.
        \item $\sqrt{d_k}$ scales it down so gradients don't explode.
        \item Softmax converts to weights (summing to 1).
        \item Result is multiplied by $V$ to get the final representation.
    \end{itemize}
\end{frame}

% --- PART 7: ISA-TF ARCHITECTURE ---

\begin{frame}[fragile]{7.1 The Paper's Invention: ISA-TF}
    The \textbf{Inter-Station-Aware Transformer Networks (ISA-TF)} combines the time-reading power of Transformers with the spatial map of the 3 Graphs.
    
    \begin{verbatim}
    [PAST DATA] --> (ENCODER) --> [FUSION] --> (DECODER) --> [FUTURE DATA]
    Last 60 mins   Finds Time   Mixes Time     Guesses the   Next 60 mins
                   Patterns     with 3 Graphs    Future
    \end{verbatim}

    \begin{itemize}
        \item \textbf{Encoder:} Uses self-attention to capture time patterns.
        \item \textbf{Fusion:} Fuses encoder output with the 3 metro graphs.
        \item \textbf{Decoder:} Predicts the future auto-regressively.
    \end{itemize}
\end{frame}

\begin{frame}{7.2 Deep Dive: Encoder \& Feed-Forward}
    \textbf{Encoder:}
    \begin{itemize}
        \item Adds "Positional Encoding" so the Transformer knows the order of time (since it doesn't read left-to-right like an RNN).
        \item Uses Self-Attention to find patterns.
        \item Passes data through a \textbf{Feed-Forward Neural Network (FFN)}:
        \[
        FFN(x) = \max(0, xW_1 + b_1)W_2 + b_2
        \]
        \textit{Notice the $\max(0, \dots)$? That is the ReLU activation function preventing vanishing gradients!}
    \end{itemize}
\end{frame}

\begin{frame}{7.3 Deep Dive: Fusion \& Decoder}
    \textbf{Intermediate Fusion (Concatenation):}
    \begin{itemize}
        \item It simply glues vectors together. 
        \item Example: $\text{Time Vector } [0.5, 0.2]$ + $\text{Graph Vector } [0.7, 0.1]$ = $\text{Fused Vector } [0.5, 0.2, 0.7, 0.1]$.
        \item Now the AI understands both TIME and SPACE simultaneously.
    \end{itemize}

    \vspace{0.2cm}
    \textbf{Decoder:}
    \begin{itemize}
        \item Uses \textbf{Masked} Self-Attention (so it can't "cheat" and look at future answers during training).
        \item Uses Cross-Attention to look back at the Fused Vector.
        \item Predicts the next 15 minutes, feeds that prediction back into itself, predicts the next 15 minutes, etc. (Auto-regressive).
    \end{itemize}
\end{frame}

% --- PART 8: EVALUATION & BASELINES ---

\begin{frame}{8.1 Evaluation Metrics (RMSE \& MAE)}
    \textbf{RMSE (Root Mean Square Error):}
    Heavily punishes large errors because differences are squared before averaging. Good for preventing massive congestion mistakes.
    $$RMSE=\sqrt{\frac{1}{n}\sum_{i=1}^{n}(\hat{D}_{i}-D_{i})^{2}}$$ 

    \textbf{MAE (Mean Absolute Error):}
    The simple average of how far off the AI's guesses were.
    $$MAE=\frac{1}{n}\sum_{i=1}^{n}|\hat{D}_{i}-D_{i}|$$ 
    
    \textit{Note: $\hat{D}_{i}$ is predicted ridership, $D_{i}$ is actual ground-truth.}
\end{frame}

\begin{frame}{8.2 Decoding the "Baselines" (Competitors)}
    The paper compares ISA-TF against older models:
    \begin{itemize}
        \item \textbf{HM (Historical Mean):} Just averages past data.
        \item \textbf{GB (Gradient Boosting):} A classic ML method combining decision trees.
        \item \textbf{RNN-GRU:} A recurrent network that processes step-by-step (forgets long-term trends).
        \item \textbf{DCRNN / PVCGN:} Advanced models that rely on \textit{Convolutions} and graph networks.
    \end{itemize}
\end{frame}

\begin{frame}{8.3 Why do "Parameters" Matter?}
    The ISA-TF model uses $\sim$47.6 Million parameters. The previous best model (PVCGN) required $\sim$134 Million.
    
    \begin{itemize}
        \item Think of parameters as the "gears" inside the AI's engine. 
        \item \textbf{Why fewer is better:} A model with fewer parameters trains faster, uses less RAM/electricity, and is much easier for a city to deploy in the real world while getting the exact same or better accuracy!
    \end{itemize}
\end{frame}

\begin{frame}{8.4 Baseline Concept: What is Convolution?}
    Many baseline models (like DCRNN) use Convolutions instead of Attention.
    
    \textbf{Definition:} Convolution slides a small mathematical filter (kernel) over data to extract local patterns.
    \begin{itemize}
        \item Example Input: $[2, 4, 6, 8]$
        \item Filter: $[1, 0]$
        \item It scans: $(2 \cdot 1 + 4 \cdot 0) = 2 \rightarrow (4 \cdot 1 + 6 \cdot 0) = 4 \dots$
    \end{itemize}
    
    While Convolutions are great for finding local spikes (like an edge in an image), they struggle to connect Station A to Station Z simultaneously, which is why the Transformer's Global Attention usually wins for long-range tasks.
\end{frame}

% --- PART 9: PAPER SPECIFIC JUSTIFICATIONS ---

\begin{frame}{9.1 Equation-Level Mapping for Slide [5/12]}
    \small
    \textbf{The paper defines three normalized edge-weight matrices (Eq. 1--3):}
    \begin{itemize}
        \item \textbf{Physical topology graph}
        \[
        M_{p}(a,b)=\frac{E(a,b)}{\sum_{k=1}^{N}E(a,k)}
        \]
        \item \textbf{Semantic similarity graph (DTW-based)}
        \[
        M_{s}(a,b)=\frac{F_{s}(a,b)}{\sum_{k=1}^{N}F_{s}(a,k)}
        \]
        \item \textbf{Correlation graph (Origin–Destination travel interactions)}
        \[
        M_{c}(a,b)=\frac{R_{c}(a,b)}{\sum_{k=1}^{N}R_{c}(a,k)}
        \]
    \end{itemize}
    \vspace{0.2cm}
    \footnotesize Interpretation: ISA-TF captures both nearby-station effects and cross-station flow interactions beyond direct physical links.
\end{frame}

\begin{frame}{9.2 Deep Dive: Physical Topology Graph ($M_p$)}
    \textbf{The Concept: Geographical Connections}
    \begin{itemize}
        \item This graph represents the actual, physical train tracks.
        \item If Station A and Station B are next to each other on the same line, they get a base score of 1 ($E(a,b)=1$).
        \item If they are not directly connected, they get a 0.
    \end{itemize}

    \vspace{0.3cm}
    \textbf{Why does the AI need this?}
    \begin{itemize}
        \item \textbf{Spillover Effect:} Passenger flow naturally moves down the line. If Station A gets extremely crowded at 8:00 AM, the AI uses this graph to predict that the next physical station down the track will likely see a surge a few minutes later.
    \end{itemize}
\end{frame}

\begin{frame}{9.3 Deep Dive: Semantic Similarity Graph ($M_s$)}
    \textbf{The Concept: Behavioral Connections}
    \begin{itemize}
        \item This graph ignores geography and looks at \textbf{station function}. 
        \item Example: Two business-district stations on opposite sides of the city might both experience massive outflow at 9:00 AM and massive inflow at 5:00 PM. 
    \end{itemize}

    \vspace{0.3cm}
    \textbf{How it's calculated: Dynamic Time Warping (DTW)}
    \begin{itemize}
        \item DTW is an algorithm that compares the shape of two time-series curves.
        \item Even if Station A peaks at 5:00 PM and Station B peaks at 5:15 PM, DTW recognizes they share the same "vibe" or pattern and links them mathematically.
        \item \textbf{Why it matters:} It helps the AI learn city-wide demographic habits.
    \end{itemize}
\end{frame}

\begin{frame}{9.4 Deep Dive: Correlation Graph ($M_c$)}
    \textbf{The Concept: Actual Human Movement}
    \begin{itemize}
        \item This graph tracks the \textbf{Origin-Destination (OD)} pairs.
        \item It simply counts how many historical passengers tapped their ticket to enter at Station A, and tapped out to exit at Station B.
    \end{itemize}

    \vspace{0.3cm}
    \textbf{Why does the AI need this?}
    \begin{itemize}
        \item The Physical Graph shows tracks, and the Semantic Graph shows schedules, but the Correlation Graph shows \textbf{actual travel paths}.
        \item If historical data shows that 40\% of people who leave a stadium at Station A travel directly to a transfer hub at Station B, this graph gives the AI the exact mathematical percentage it needs to predict the crowd at Station B.
    \end{itemize}
\end{frame}
